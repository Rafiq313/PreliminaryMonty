cmake_minimum_required(VERSION 3.20)
project(PreliminaryMonty VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(SDL2 CONFIG REQUIRED)
find_package(SQLite3 QUIET)
if(NOT SQLite3_FOUND)
  find_library(MONTY_SQLITE_LIBRARY NAMES sqlite3 libsqlite3.so.0 REQUIRED)
  add_library(SQLite::SQLite3 UNKNOWN IMPORTED)
  set_target_properties(SQLite::SQLite3 PROPERTIES IMPORTED_LOCATION "${MONTY_SQLITE_LIBRARY}")
endif()
add_library(monty_core STATIC src/maze.c src/assets.c src/audio.c src/game.c src/input.c src/video.c src/room.c)
target_include_directories(monty_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(monty_core PUBLIC SQLite::SQLite3)
if(MSVC)
  target_compile_options(monty_core PRIVATE /W4)
else()
  target_compile_options(monty_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(preliminary_monty src/main.c src/paths.c src/frontend_sdl.c)
target_link_libraries(preliminary_monty PRIVATE monty_core SDL2::SDL2)
